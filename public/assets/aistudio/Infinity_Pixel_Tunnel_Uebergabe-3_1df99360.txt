package at.hausgarten.service

import android.annotation.SuppressLint
import android.content.Context
import android.content.SharedPreferences
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import android.util.Base64
import java.io.IOException
import java.security.KeyStore
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

/** Keystore keys stay on Android; runtime secrets never enter the UI data bridge. */
class PixelTunnelConfig(context: Context) {
    private val prefs: SharedPreferences = context.getSharedPreferences(PREFS, Context.MODE_PRIVATE)

    private fun key(): SecretKey {
        val store = KeyStore.getInstance("AndroidKeyStore")
        store.load(null)
        if (!store.containsAlias(ALIAS)) {
            val generator =
                KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore")
            generator.init(
                KeyGenParameterSpec.Builder(
                        ALIAS,
                        KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT,
                    )
                    .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                    .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                    .build()
            )
            generator.generateKey()
        }
        return store.getKey(ALIAS, null) as SecretKey
    }

    @Synchronized
    @Throws(Exception::class)
    fun save(id: String, secret: String?) {
        save(id, organization(), secret)
    }

    @Synchronized
    @Throws(Exception::class)
    fun save(id: String, organization: String?, secret: String?) {
        validateId(id)
        validateOrganization(organization)
        if (
            secret == null ||
                secret.length < 20 ||
                secret.length > 4096 ||
                secret.matches(Regex("(?s).*\\s.*"))
        ) {
            throw IllegalArgumentException("Bitte den Runtime-Schlüssel vollständig eingeben.")
        }
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.ENCRYPT_MODE, key())
        val ciphertext = cipher.doFinal(secret.toByteArray(Charsets.UTF_8))
        if (
            !prefs
                .edit()
                .putString("id", id)
                .putString("organization", organization)
                .putString("iv", Base64.encodeToString(cipher.iv, Base64.NO_WRAP))
                .putString("secret", Base64.encodeToString(ciphertext, Base64.NO_WRAP))
                .commit()
        ) {
            throw IOException("Die Verbindungseinstellungen konnten nicht gespeichert werden.")
        }
    }

    fun id(): String = prefs.getString("id", "") ?: ""

    fun organization(): String = prefs.getString("organization", "") ?: ""

    fun organization(value: String?) {
        validateOrganization(value)
        if (!prefs.edit().putString("organization", value).commit())
            throw IllegalStateException("Die Organisationsangabe konnte nicht gespeichert werden.")
    }

    fun configured(): Boolean =
        id().isNotEmpty() && !(prefs.getString("secret", "") ?: "").isEmpty()

    @Throws(Exception::class)
    fun secret(): String {
        validateId(id())
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(
            Cipher.DECRYPT_MODE,
            key(),
            GCMParameterSpec(128, Base64.decode(prefs.getString("iv", ""), Base64.NO_WRAP)),
        )
        return String(
            cipher.doFinal(Base64.decode(prefs.getString("secret", ""), Base64.NO_WRAP)),
            Charsets.UTF_8,
        )
    }

    fun enabled(): Boolean = prefs.getBoolean("enabled", false)

    fun writeEnabled(): Boolean = prefs.getBoolean("worklog_write_enabled", false)

    fun writeEnabled(value: Boolean) {
        if (!prefs.edit().putBoolean("worklog_write_enabled", value).commit())
            throw IllegalStateException("Die Schreibfreigabe konnte nicht gespeichert werden.")
    }

    @SuppressLint("ApplySharedPref")
    fun enabled(value: Boolean) {
        if (!prefs.edit().putBoolean("enabled", value).commit())
            throw IllegalStateException("Die Verbindungsfreigabe konnte nicht gespeichert werden.")
    }

    companion object {
        private const val ALIAS = "hausgarten_pixel_tunnel_v1"
        private const val PREFS = "pixel_tunnel_private"

        @JvmStatic
        fun validateId(id: String) {
            if (!id.matches(Regex("tunnel_(?:[a-z0-9]{4}_)?[a-z0-9]{32}")))
                throw IllegalArgumentException("Bitte die vollständige private Tunnel-ID eingeben.")
        }

        @JvmStatic
        fun validateOrganization(organization: String?) {
            if (
                organization == null ||
                    organization.isNotEmpty() &&
                        !organization.matches(Regex("org-[a-zA-Z0-9]{4,64}"))
            )
                throw IllegalArgumentException(
                    "Bitte die OpenAI-Organisations-ID im Format org-… eingeben."
                )
        }
    }
}

package at.hausgarten.service

import java.io.*
import java.net.*
import java.nio.charset.StandardCharsets
import java.util.*
import org.json.JSONArray
import org.json.JSONObject

class PixelTunnelWire {
    fun interface Transport {
        @Throws(Exception::class)
        fun request(
            method: String,
            path: String,
            body: JSONObject?,
            shard: String?,
            timeoutMs: Int,
        ): Reply
    }

    class Reply {
        @JvmField val code: Int
        @JvmField val json: JSONObject
        @JvmField val receivedNanos: Long

        constructor(code: Int, json: JSONObject) : this(code, json, System.nanoTime()) {}

        constructor(code: Int, json: JSONObject, time: Long) {
            this.code = code
            this.json = json
            this.receivedNanos = time
        }
    }

    class AccessError : IOException {
        @JvmField val httpCode: Int
        @JvmField val reason: String

        constructor(code: Int) : this(code, "auth_denied") {}

        constructor(
            code: Int,
            reason: String,
        ) : super("Tunnel-ID oder Runtime-Berechtigung prüfen.") {
            httpCode = code
            this.reason = reason
        }

        fun userMessage(): String {
            when (reason) {
                "organization_required" -> {
                    return "Die OpenAI-Organisationsangabe fehlt. Verbindungseinstellungen öffnen und die Organisation des Pixel-Tunnels bestätigen."
                }
                "organization_mismatch" -> {
                    return "Der gespeicherte Schlüssel gehört zu einer anderen OpenAI-Organisation als der Pixel-Tunnel. Bitte einen Schlüssel aus der Organisation des Tunnels eingeben."
                }
                "tunnel_access_denied" -> {
                    return "Der Schlüssel darf diesen Tunnel nicht verwenden. Im richtigen OpenAI-Konto Tunnels Read + Use prüfen."
                }
                "invalid_api_key" -> {
                    return "OpenAI erkennt den gespeicherten Schlüssel nicht. Bitte den gültigen Runtime-Schlüssel erneut im privaten Einrichtungsfenster eingeben."
                }
                else -> {
                    return (("OpenAI verweigert die Tunnel-Anmeldung (HTTP " + httpCode) +
                        "). Organisation, Schlüssel und Tunnels Read + Use prüfen.")
                }
            }
        }
    }

    class RateLimit : IOException {
        @JvmField val retryAfterMs: Long

        constructor(wait: Long) : super("Tunnel-Anfragen sind vorübergehend begrenzt.") {
            retryAfterMs = Math.max(1000, Math.min(900000, wait))
        }
    }

    private val id: String
    private val transport: Transport
    private val tools: PixelMcpReadTools

    private class Cached {
        @JvmField val fingerprint: String
        @JvmField val response: JSONObject

        constructor(fingerprint: String, response: JSONObject) {
            this.fingerprint = fingerprint
            this.response = response
        }
    }

    private val results: MutableMap<String, Cached> =
        object : LinkedHashMap<String, Cached>() {
            override fun removeEldestEntry(e: MutableMap.MutableEntry<String, Cached>?): Boolean {
                return (size > 128)
            }
        }

    constructor(id: String, transport: Transport, tools: PixelMcpReadTools) {
        PixelTunnelConfig.validateId(id)
        this.id = id
        this.transport = transport
        this.tools = tools
    }

    private fun path(suffix: String): String {
        return (("/v1/tunnels/" + id) + suffix)
    }

    @Throws(Exception::class)
    fun verify(): Unit {
        check(transport.request("GET", path(""), null, null, 15000))
    }

    @Throws(Exception::class)
    fun cycle(): Int {
        var poll =
            transport.request("GET", path("/poll?limit=1&timeout_ms=15000"), null, null, 35000)
        check(poll)
        if (poll.code == 204) {
            return 0
        }
        val commands = poll.json.getJSONArray("commands")
        var delivered: Int = 0
        for (i in 0 until commands.length()) {
            if (Thread.currentThread().isInterrupted()) {
                throw InterruptedException()
            }
            val command = commands.getJSONObject(i)
            val expiry: Long = deadline(command.opt("response_timeout"), poll.receivedNanos)
            if ((expiry != Long.MAX_VALUE) && (System.nanoTime() >= expiry)) {
                continue
            }
            val requestId = command.getString("request_id")
            val shard = command.getString("shard_token")
            if (
                (((((requestId.isEmpty() || (requestId.length > 4096)) || shard.isEmpty()) ||
                    (shard.length > 16384)) || (shard.indexOf('\r') >= 0)) ||
                    (shard.indexOf('\n') >= 0))
            ) {
                throw IOException("Ungültige Tunnel-Antwort.")
            }
            val type = command.optString("command_type")
            val channel = command.optString("channel", "main")
            if (!("jsonrpc" == type) && !("session_termination" == type)) {
                continue
            }
            var fingerprint =
                ((((type + "\n") + channel) + "\n") + canonical(command.optJSONObject("jsonrpc")))
            val cached = results.get(requestId)
            if ((cached != null) && !(cached.fingerprint == fingerprint)) {
                throw IOException("Widersprüchliche Anfrage-ID. Verbindung wird erneut geprüft.")
            }
            var response = (if (cached == null) null else cached.response)
            if (response == null) {
                response = JSONObject().put("request_id", requestId).put("channel", channel)
                if (type == "session_termination") {
                    response.put("resp_code", 204).put("resp_type", "session_termination_response")
                } else {
                    val rpc = command.getJSONObject("jsonrpc")
                    var result: JSONObject?
                    if (!("main" == channel)) {
                        result =
                            JSONObject()
                                .put("jsonrpc", "2.0")
                                .put(
                                    "id",
                                    (if (rpc.opt("id") == null) JSONObject.NULL else rpc.opt("id")),
                                )
                                .put(
                                    "error",
                                    JSONObject()
                                        .put("code", -(32601))
                                        .put("message", "Channel nicht unterstützt."),
                                )
                    } else {
                        result = tools.rpc(rpc)
                    }
                    response
                        .put("resp_code", (if (result == null) 202 else 200))
                        .put(
                            "resp_type",
                            (if (result == null) "notify_ack" else "jsonrpc_response"),
                        )
                    if (result != null) {
                        response
                            .put("resp_json", result)
                            .put(
                                "resp_headers",
                                JSONObject()
                                    .put("Content-Type", JSONArray().put("application/json")),
                            )
                    }
                }
                results.put(requestId, Cached(fingerprint, response))
            }
            if ((expiry != Long.MAX_VALUE) && (System.nanoTime() >= expiry)) {
                continue
            }
            var timeout: Int =
                (if (expiry == Long.MAX_VALUE) 15000
                else
                    (Math.max(1, Math.min(15000, ((expiry - System.nanoTime()) / 1000000))))
                        .toInt())
            check(transport.request("POST", path("/response"), response, shard, timeout))
            delivered++
        }
        return delivered
    }

    class HttpTransport : Transport {
        private val root: String
        private val secret: String
        private val organization: String
        private val instance: String = UUID.randomUUID().toString()
        @Volatile private var active: HttpURLConnection? = null

        constructor(secret: String) : this("https://api.openai.com", secret, false) {}

        constructor(
            secret: String,
            organization: String,
        ) : this("https://api.openai.com", secret, false, organization) {}

        constructor(root: String, secret: String, test: Boolean) : this(root, secret, test, "") {}

        constructor(root: String, secret: String, test: Boolean, organization: String) {
            if (
                (!(root == "https://api.openai.com") &&
                    !(test && root.matches(Regex("http://127\\.0\\.0\\.1:[0-9]{1,5}"))))
            ) {
                throw IllegalArgumentException("Nicht erlaubtes Tunnel-Ziel.")
            }
            PixelTunnelConfig.validateOrganization(organization)
            this.root = root
            this.secret = secret
            this.organization = organization
        }

        fun close(): Unit {
            var current = active
            if (current != null) {
                current.disconnect()
            }
        }

        override @Throws(Exception::class) fun request(
            method: String,
            path: String,
            body: JSONObject?,
            shard: String?,
            timeoutMs: Int,
        ): Reply {
            if (!(path.startsWith("/v1/tunnels/")) || path.contains("..")) {
                throw IOException("Ungültiger Tunnel-Pfad.")
            }
            val connection = (URL(root + path).openConnection() as HttpURLConnection)
            active = connection
            try {
                connection.setInstanceFollowRedirects(false)
                connection.setConnectTimeout(Math.min(timeoutMs, 15000))
                connection.setReadTimeout(timeoutMs)
                connection.setRequestMethod(method)
                connection.setRequestProperty("Authorization", ("Bearer " + secret))
                connection.setRequestProperty("Accept", "application/json")
                if (!(organization.isEmpty())) {
                    connection.setRequestProperty("OpenAI-Organization", organization)
                }
                connection.setRequestProperty("X-Tunnel-Client-Name", "hausgarten-android")
                connection.setRequestProperty("X-Tunnel-Client-Version", "0.1.0")
                connection.setRequestProperty("X-Tunnel-Client-Instance-Id", instance)
                connection.setRequestProperty(
                    "X-Tunnel-MCP-Server-Info",
                    "{\"version\":1,\"channels\":[{\"name\":\"main\",\"proc_affinity\":true}]}",
                )
                if (shard != null) {
                    connection.setRequestProperty("X-Tunnel-Shard-Token", shard)
                }
                if (body != null) {
                    connection.setDoOutput(true)
                    connection.setRequestProperty("Content-Type", "application/json")
                    val bytes = body.toString().toByteArray(StandardCharsets.UTF_8)
                    connection.setFixedLengthStreamingMode(bytes.size)
                    connection.outputStream.use { out -> out.write(bytes) }
                }
                val code: Int = connection.responseCode
                val receipt: Long = System.nanoTime()
                if (code == 204) {
                    return Reply(code, JSONObject(), receipt)
                }
                if ((code < 200) || (code >= 300)) {
                    val safe = JSONObject()
                    if (code == 429) {
                        safe.put(
                            "retry_after_ms",
                            retryAfter(
                                connection.getHeaderField("Retry-After"),
                                System.currentTimeMillis(),
                            ),
                        )
                    }
                    if ((code == 401) || (code == 403)) {
                        safe.put("access_reason", readAccessReason(connection))
                    }
                    return Reply(code, safe, receipt)
                }
                connection.inputStream.use { input ->
                    ByteArrayOutputStream().use { out ->
                        val buffer = ByteArray(8192)
                        var count: Int
                        var total: Int = 0
                        while (true) {
                            count = input.read(buffer)
                            if (count == -1) break
                            total += count
                            if (total > ((2 * 1024) * 1024)) {
                                throw IOException("Tunnel-Antwort zu groß.")
                            }
                            out.write(buffer, 0, count)
                        }
                        var value = String(out.toByteArray(), StandardCharsets.UTF_8)
                        return Reply(
                            code,
                            (if (value.trim().isEmpty()) JSONObject() else JSONObject(value)),
                            receipt,
                        )
                    }
                }
            } finally {
                connection.disconnect()
                if (active == connection) {
                    active = null
                }
            }
        }

        companion object {
            private @JvmStatic fun readAccessReason(connection: HttpURLConnection): String {
                try {
                    connection.errorStream.use { input ->
                        ByteArrayOutputStream().use { out ->
                            if (input == null) {
                                return "auth_denied"
                            }
                            val buffer = ByteArray(4096)
                            var count: Int
                            var total: Int = 0
                            while (true) {
                                count = input.read(buffer)
                                if (count == -1) break
                                total += count
                                if (total > 65536) {
                                    return "auth_denied"
                                }
                                out.write(buffer, 0, count)
                            }
                            return accessReason(
                                JSONObject(String(out.toByteArray(), StandardCharsets.UTF_8))
                            )
                        }
                    }
                } catch (ignored: Exception) {
                    return "auth_denied"
                }
            }
        }
    }

    companion object {
        @JvmStatic
        fun retryAfter(value: String?, now: Long): Long {
            if (value == null) {
                return 15000
            }
            try {
                return Math.max(
                    1000,
                    Math.min(
                        900000,
                        Math.multiplyExact(java.lang.Long.parseLong(value.trim()), 1000L),
                    ),
                )
            } catch (ignored: Exception) {
                try {
                    return Math.max(
                        1000,
                        Math.min(
                            900000,
                            (java.time.ZonedDateTime.parse(
                                    value,
                                    java.time.format.DateTimeFormatter.RFC_1123_DATE_TIME,
                                )
                                .toInstant()
                                .toEpochMilli() - now),
                        ),
                    )
                } catch (invalid: Exception) {
                    return 15000
                }
            }
        }

        private @JvmStatic @Throws(Exception::class) fun check(r: Reply): Unit {
            if (r.code == 429) {
                throw RateLimit(r.json.optLong("retry_after_ms", 15000))
            }
            if ((r.code == 401) || (r.code == 403)) {
                throw AccessError(r.code, r.json.optString("access_reason", "auth_denied"))
            }
            if ((r.code < 200) || (r.code >= 300)) {
                throw IOException(
                    (("Tunnel vorübergehend nicht erreichbar (HTTP " + r.code) + ").")
                )
            }
        }

        @JvmStatic
        fun deadline(timeout: Any?, receipt: Long): Long {
            if (
                (!(timeout is String) ||
                    !((timeout as String).matches(Regex("[0-9]+(?:ns|us|ms|s|m|h)"))))
            ) {
                return Long.MAX_VALUE
            }
            val m = java.util.regex.Pattern.compile("([0-9]+)([a-z]+)").matcher(timeout as String)
            m.matches()
            try {
                var amount: Long = java.lang.Long.parseLong(m.group(1)!!)
                var factor: Long
                when (m.group(2)) {
                    "ns" -> {
                        factor = 1
                    }
                    "us" -> {
                        factor = 1000
                    }
                    "ms" -> {
                        factor = 1000000
                    }
                    "s" -> {
                        factor = 1000000000
                    }
                    "m" -> {
                        factor = 60000000000L
                    }
                    else -> {
                        factor = 3600000000000L
                    }
                }
                return Math.addExact(receipt, Math.multiplyExact(amount, factor))
            } catch (invalid: Exception) {
                return Long.MAX_VALUE
            }
        }

        private @JvmStatic @Throws(Exception::class) fun canonical(value: Any?): String {
            if (value is JSONObject) {
                val obj = (value as JSONObject)
                val keys = ArrayList<String>()
                val iterator = obj.keys()
                while (iterator.hasNext()) {
                    keys.add(iterator.next())
                }
                Collections.sort(keys)
                var out = StringBuilder("{")
                for (key in keys) {
                    out.append(JSONObject.quote(key))
                        .append(':')
                        .append(canonical(obj.get(key)))
                        .append(',')
                }
                return out.append('}').toString()
            }
            if (value is JSONArray) {
                val array = (value as JSONArray)
                var out = StringBuilder("[")
                for (i in 0 until array.length()) {
                    out.append(canonical(array.get(i))).append(',')
                }
                return out.append(']').toString()
            }
            return (if ((value == null) || (value == JSONObject.NULL)) "null"
            else (if (value is String) JSONObject.quote(value as String) else value.toString()))
        }

        @JvmStatic
        fun accessReason(json: JSONObject): String {
            var error = json.optJSONObject("error")
            var message =
                ((((java.lang.String.valueOf(
                    (if (error == null) "" else error.optString("message"))
                ) + " ") + json.optString("detail")) + " ") + json.optString("message"))
            if (json.opt("error") is String) {
                message += (" " + json.optString("error"))
            }
            message = message.lowercase(Locale.ROOT)
            if (
                message.contains("openai-organization header should match organization for api key")
            ) {
                return "organization_mismatch"
            }
            if (message.contains("requires an active organization context")) {
                return "organization_required"
            }
            if (message.contains("does not have permission to use this tunnel")) {
                return "tunnel_access_denied"
            }
            if ((error != null) && ("invalid_api_key" == error.optString("code"))) {
                return "invalid_api_key"
            }
            return "auth_denied"
        }
    }
}

    private fun runTunnel(localStore: PixelLocalStore) {
        var failures = 0
        while (!closing.get()) {
            try {
                val config = PixelTunnelConfig(this)
                if (!config.enabled()) break
                val binding = connectionBinding(config)
                activeBinding = binding
                if (!online()) {
                    status("offline", "Kein Internet. Verbindung wird automatisch erneut versucht.")
                    Thread.sleep(5000)
                    continue
                }
                readState()
                val transport =
                    PixelTunnelWire.HttpTransport(config.secret(), config.organization())
                http = transport
                val allowed =
                    if (config.writeEnabled())
                        object : PixelMcpWriteTools.Sink {
                            override fun writable(): Boolean =
                                PixelTunnelConfig(this@PixelTunnelService).writeEnabled()

                            override fun execute(tool: String, args: JSONObject): JSONObject {
                                if (!PixelTunnelConfig(this@PixelTunnelService).writeEnabled())
                                    throw PixelMcpWriteTools.Rejected(
                                        "WRITE_DISABLED",
                                        "Schreibzugriff wurde beendet.",
                                    )
                                return localStore.execute(tool, args)
                            }

                            override fun status(id: String, state: JSONObject): JSONObject =
                                localStore.status(id, state)
                        }
                    else null
                val tunnel =
                    PixelTunnelWire(
                        config.id(),
                        transport,
                        PixelMcpReadTools(
                            localStore.readSource(),
                            allowed,
                            PixelOwnerAccess(this, localStore),
                        ),
                    )
                status("connecting", "OpenAI-Tunnel wird geprüft.")
                tunnel.verify()
                failures = 0
                while (!closing.get() && config.enabled() && connectionBinding(config) == binding) {
                    if (config.writeEnabled() != (allowed != null)) break
                    val delivered = tunnel.cycle()
                    status(
                        "ready",
                        if (delivered > 0) "Anfrageantwort an den Tunnel übergeben."
                        else "Tunnel erreichbar · wartet auf Anfragen.",
                        contact = true,
                        response = delivered > 0,
                    )
                }
            } catch (limited: PixelTunnelWire.RateLimit) {
                failures = minOf(8, failures + 1)
                status(
                    "rate_limited",
                    "Tunnel-Anfragen vorübergehend begrenzt. Nach der Wartezeit wird erneut verbunden.",
                )
                try {
                    Thread.sleep(maxOf(limited.retryAfterMs, minOf(60000L, 1000L shl failures)))
                } catch (_: InterruptedException) {
                    Thread.currentThread().interrupt()
                    break
                }
            } catch (denied: PixelTunnelWire.AccessError) {
                status("auth_error", denied.userMessage())
                val current = PixelTunnelConfig(this)
                if (connectionBinding(current) == activeBinding) current.enabled(false)
                break
            } catch (_: InterruptedException) {
                Thread.currentThread().interrupt()
                break
            } catch (_: Exception) {
                failures++
                status(
                    "reconnecting",
                    "Verbindung oder lokalen Datenstand prüfen. Neuer Versuch folgt.",
                )
                try {
                    Thread.sleep(minOf(60000L, 1000L shl minOf(failures, 6)))
                } catch (_: InterruptedException) {
                    Thread.currentThread().interrupt()
                    break
                }
            } finally {
                http?.close()
            }
        }
    }

package at.hausgarten.service

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import java.util.concurrent.Executors

/** Reboot/update recovery after explicit opt-in and complete configuration. */
class PixelTunnelBootReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        if (
            intent.action != Intent.ACTION_BOOT_COMPLETED &&
                intent.action != Intent.ACTION_MY_PACKAGE_REPLACED
        )
            return
        val pending = goAsync()
        executor.execute {
            try {
                // Broadcasts have a short completion deadline; a protected restore resumes in-app.
                try {
                    RevierApplication.awaitStartup(context, 5000)
                } catch (_: RuntimeException) {
                    return@execute
                }
                if (PixelTunnelService.recoveryStartBlocked(context)) return@execute
                val config = PixelTunnelConfig(context)
                if (!config.enabled() || !config.configured()) return@execute
                try {
                    context.startForegroundService(Intent(context, PixelTunnelService::class.java))
                } catch (_: RuntimeException) {
                    context
                        .getSharedPreferences("pixel_tunnel_status", Context.MODE_PRIVATE)
                        .edit()
                        .putString("state", "start_blocked")
                        .putString(
                            "detail",
                            "Android hat den Start eingeschränkt. Bitte Verbindung in der App starten.",
                        )
                        .apply()
                }
            } finally {
                pending.finish()
            }
        }
    }

    companion object {
        private val executor =
            Executors.newSingleThreadExecutor { task ->
                Thread(task, "pixel-tunnel-boot").apply { isDaemon = true }
            }
    }

/** Native credential setup; secrets stay outside saved state and business data. */
class PixelTunnelActivity : ComponentActivity() {
    private lateinit var config: PixelTunnelConfig
    private var status by mutableStateOf("Status wird geprüft …")
    private var battery by mutableStateOf("")
    private var diagnosis by mutableStateOf("")
    private var lastContact by mutableStateOf("Noch kein bestätigter Tunnelkontakt.")
    private var lastResponse by mutableStateOf("Noch keine beantwortete Anfrage bestätigt.")
    private var access by mutableStateOf("")

    private var runtimeKey by mutableStateOf("")

    override fun onCreate(savedInstanceState: Bundle?) {

                val initial =

                            config.enabled().toString(),

                organization = initial[1]
                enabled = initial[2].toBoolean()
                write = requestedWrite ?: initial[3].toBoolean()
                configured = initial[4].toBoolean()
                refreshStatus()
                configurationReady = true
                busy = false
                consumeNotificationResult()

                throw cancelled

    override fun onResume() {
        super.onResume()
        if (configurationReady) lifecycleScope.launch { refreshStatus() }
    }

    override fun onSaveInstanceState(outState: Bundle) {
        super.onSaveInstanceState(outState)
        requestedWrite?.let { outState.putBoolean("pending_notification_write", it) }
        pendingNotificationResult?.let { outState.putBoolean("pending_notification_result", it) }
    }

    private fun open(action: String, uri: Uri?) {
        try {
            startActivity(Intent(action).apply { if (uri != null) data = uri })
        } catch (_: ActivityNotFoundException) {
            toast("Diese Einstellung ist auf dem Gerät nicht verfügbar.")
        }
    }

    private fun startConnection() {
        if (busy) return
        busy = true
        val entered = runtimeKey.trim()
        val tunnel = tunnelId.trim()
        val org = organization.trim()
        lifecycleScope.launch {
            try {
                withContext(Dispatchers.IO) {
                    PixelTunnelConfig.validateId(tunnel)
                    PixelTunnelConfig.validateOrganization(org)
                    if (entered.isNotEmpty()) {
                        stopService(
                            Intent(this@PixelTunnelActivity, PixelTunnelService::class.java)
                        )
                        config.save(tunnel, org, entered)
                    } else if (!config.configured() || config.id() != tunnel)
                        throw IllegalArgumentException(
                            "Bitte für den neuen Tunnel den Runtime-Schlüssel eingeben."
                        )
                    else if (config.organization() != org) {
                        stopService(
                            Intent(this@PixelTunnelActivity, PixelTunnelService::class.java)
                        )
                        config.organization(org)
                    }
                }
                if (entered.isNotEmpty()) runtimeKey = ""
                configured = true
                requestConnection(write)
            } catch (error: Exception) {
                AlertDialog.Builder(this@PixelTunnelActivity)
                    .setTitle("Verbindung noch nicht eingerichtet")
                    .setMessage(
                        if (error is IllegalArgumentException) error.message
                        else
                            "Einstellungen konnten nicht sicher gespeichert werden. Bitte erneut versuchen."
                    )
                    .setPositiveButton("OK", null)
                    .show()
            } finally {
                busy = false
            }
        }
    }

    /**
     * Takes effect at once: a running connection restarts with the new access, otherwise only the
     * choice is saved.
     */
    private fun changeWrite(permitted: Boolean) {
        if (busy) return
        write = permitted
        busy = true
        lifecycleScope.launch {
            try {
                if (enabled && configured) requestConnection(permitted)
                else {
                    withContext(Dispatchers.IO) { config.writeEnabled(permitted) }
                    refreshStatus()
                }
            } finally {
                busy = false
            }
        }
    }

    private fun startSavedConnection() {
        if (busy) return
        busy = true
        lifecycleScope.launch {
            try {
                requestConnection(withContext(Dispatchers.IO) { config.writeEnabled() })
            } finally {
                busy = false
            }
        }
    }

    private suspend fun requestConnection(permitted: Boolean) {
        pendingNotificationResult = null
        if (
            Build.VERSION.SDK_INT >= 33 &&
                checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) !=
                    PackageManager.PERMISSION_GRANTED
        ) {
            requestedWrite = permitted
            try {
                notificationPermission.launch(Manifest.permission.POST_NOTIFICATIONS)
            } catch (failure: RuntimeException) {
                requestedWrite = null
                throw failure
            }
        } else {
            requestedWrite = null
            launchConnection(permitted)
        }
    }

    private suspend fun launchConnection(permitted: Boolean) {
        try {
            withContext(Dispatchers.IO) {
                if (config.writeEnabled() != permitted)
                    stopService(Intent(this@PixelTunnelActivity, PixelTunnelService::class.java))
                config.writeEnabled(permitted)
                config.enabled(true)
                startForegroundService(
                    Intent(this@PixelTunnelActivity, PixelTunnelService::class.java)
                )
            }
            refreshStatus()
        } catch (_: RuntimeException) {
            withContext(Dispatchers.IO) { config.enabled(false) }
            enabled = false
            toast("Android hat den Dienststart eingeschränkt. Bitte die Akku-Einstellungen prüfen.")
        }
    }

    private fun stopConnection() {
        if (busy) return
        requestedWrite = null
        pendingNotificationResult = null
        lifecycleScope.launch {
            busy = true
            try {
                withContext(Dispatchers.IO) {
                    config.enabled(false)
                    stopService(Intent(this@PixelTunnelActivity, PixelTunnelService::class.java))
                }
                refreshStatus()
            } finally {
                busy = false
            }
        }
    }

    private fun consumeNotificationResult() {
        if (!configurationReady) return
        val permitted = requestedWrite ?: return
        val granted = pendingNotificationResult ?: return
        requestedWrite = null
        pendingNotificationResult = null
        if (!granted) {
            toast(
                "Bitte Benachrichtigungen freigeben, damit du die laufende Verbindung sehen und beenden kannst."
            )
            return
        }
        busy = true
        lifecycleScope.launch {
            try {
                launchConnection(permitted)
            } finally {
                busy = false
            }
        }
    }

    private suspend fun refreshStatus() {
        try {
            val info =
                withContext(Dispatchers.IO) { PixelTunnelService.status(this@PixelTunnelActivity) }
            diagnosis = info.optString("detail")
            enabled = info.optBoolean("enabled")
            configured = info.optBoolean("configured")
            status = pixelConnectionLabel(info)
            statusTone = pixelConnectionTone(info)
            lastContact =
                pixelConnectionTime(
                    info.optLong("last_contact_at"),
                    "Noch kein bestätigter Tunnelkontakt.",
                )
            lastResponse =
                pixelConnectionTime(
                    info.optLong("last_response_at"),
                    "Noch keine beantwortete Anfrage bestätigt.",
                )
            access =
                (if (enabled) "Lesen lokal eingeschaltet."
                else "Verbindung und Zugriff ausgeschaltet.") +
                    "\nGespeicherte Schreibfreigabe: " +
                    (if (info.optBoolean("write_enabled")) "EIN." else "AUS.") +
                    " Keine Schreibprobe ausgeführt."
            battery =
                if (info.optBoolean("battery_exempt")) "Akku-Ausnahme ist aktiv."
                else
                    "Akku-Ausnahme noch nicht aktiv. Bei Energiesparen kann die Verbindung unterbrochen werden."
        } catch (cancelled: kotlinx.coroutines.CancellationException) {
            throw cancelled
        } catch (_: Exception) {
            status = "Lokaler Verbindungsstatus derzeit nicht lesbar."
            statusTone = NativeStatusTone.Attention
        }
    }

    private fun toast(text: String) {
        Toast.makeText(this, text, Toast.LENGTH_LONG).show()
    }
}

internal fun pixelConnectionLabel(info: org.json.JSONObject): String =
    when {
        info.optString("state") == "status_unavailable" -> "Lokaler Status nicht lesbar"
        info.optString("state") == "auth_error" -> "Verbindungsfreigabe prüfen"
        !info.optBoolean("configured") -> "Noch nicht eingerichtet"
        !info.optBoolean("enabled") -> "Pixel-Verbindung ausgeschaltet"
        !info.optBoolean("running") -> "Tunnel eingeschaltet · Pixel-Dienst läuft nicht"
        info.optBoolean("transport_confirmed") -> "Tunnel eingeschaltet · letzter Kontakt bestätigt"
        info.optString("state") in
            listOf(
                "offline",
                "reconnecting",
                "rate_limited",
                "local_review_required",
                "start_blocked",
            ) -> "Tunnel eingeschaltet · derzeit nicht bestätigt"
        else -> "Tunnel eingeschaltet · Verbindung ausstehend"
    }

internal fun pixelConnectionTone(info: org.json.JSONObject): NativeStatusTone =
    when {
        info.optString("state") in
            listOf("auth_error", "local_review_required", "start_blocked", "status_unavailable") ->
            NativeStatusTone.Attention
        !info.optBoolean("enabled") -> NativeStatusTone.Neutral
        info.optBoolean("transport_confirmed") -> NativeStatusTone.Confirmed
        else -> NativeStatusTone.Attention
    }

internal fun pixelConnectionTime(value: Long, empty: String): String =
    if (value <= 0) empty
    else
        DateTimeFormatter.ofPattern("dd.MM.yyyy HH:mm:ss", Locale.GERMAN)
            .withZone(ZoneId.of("Europe/Vienna"))
            .format(Instant.ofEpochMilli(value))
